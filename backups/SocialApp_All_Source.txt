function formatPostTime(createdAt) {
  if (!createdAt) return '';
  const date = new Date(createdAt);
  if (Number.isNaN(date.getTime())) return '';

  const seconds = Math.max(0, Math.floor((Date.now() - date.getTime()) / 1000));
  if (seconds < 60) return 'Just now';
  if (seconds < 3600) return `${Math.floor(seconds / 60)} minutes ago`;
  if (seconds < 86400) return `${Math.floor(seconds / 3600)} hours ago`;
  if (seconds < 172800) return 'Yesterday';
  if (seconds < 604800) return `${Math.floor(seconds / 86400)} days ago`;
  return `${date.toLocaleDateString()} · ${date.toLocaleTimeString([], {
    hour: 'numeric',
    minute: '2-digit',
  })}`;
}

function PostCard({ post, onOpen = () => {} }) {
  const [liked, setLiked] = React.useState(false);
  const [likeCount, setLikeCount] = React.useState(
    Number(String(post.likes || '0').replace(/,/g, '')) || 0
  );
  const [saved, setSaved] = React.useState(false);
  const [commentsOpen, setCommentsOpen] = React.useState(false);
  const [comments, setComments] = React.useState([]);
  const [commentDraft, setCommentDraft] = React.useState('');
  const [interactionBusy, setInteractionBusy] = React.useState(false);

  const handleLike = async () => {
    if (interactionBusy) return;

    if (!post.id) {
      setLiked(previous => {
        setLikeCount(count => Math.max(0, count + (previous ? -1 : 1)));
        return !previous;
      });
      return;
    }

    setInteractionBusy(true);
    try {
      const result = await togglePostLike(post.id);
      setLiked(Boolean(result.liked));
      setLikeCount(Number(result.likes) || 0);
    } catch (error) {
      Alert.alert('Like failed', error.message || 'Could not update like.');
    } finally {
      setInteractionBusy(false);
    }
  };

  const openComments = async () => {
    setCommentsOpen(true);
    if (!post.id) {
      setComments([]);
      return;
    }

    try {
      const result = await loadPostComments(post.id);
      setComments(result);
    } catch (error) {
      Alert.alert('Comments', error.message || 'Could not load comments.');
    }
  };

  const submitComment = async () => {
    const text = commentDraft.trim();
    if (!text || interactionBusy) return;

    if (!post.id) {
      Alert.alert('Comments', 'Save this post to the server before commenting.');
      return;
    }

    setInteractionBusy(true);
    try {
      const comment = await createPostComment(post.id, text);
      setComments(previous => [...previous, comment]);
      setCommentDraft('');
    } catch (error) {
      Alert.alert('Comment failed', error.message || 'Could not post comment.');
    } finally {
      setInteractionBusy(false);
    }
  };

  return (
    <View style={styles.post}>
      <View style={styles.postHeader}>
        <View style={styles.smallAvatar}>
          {post.avatarUrl ? (
            <Image source={{ uri: post.avatarUrl }} style={{ width: 40, height: 40, borderRadius: 20 }} />
          ) : (
            <Text style={styles.smallAvatarText}>{(post.user || 'U').charAt(0).toUpperCase()}</Text>
          )}
        </View>
        <View style={styles.postUserInfo}>
          <Text style={styles.username}>{post.user || 'you'}</Text>
          {!!post.location && <Text style={styles.location}>{post.location}</Text>}
        </View>
        <TouchableOpacity><Text style={styles.more}>•••</Text></TouchableOpacity>
      </View>

      <TouchableOpacity activeOpacity={0.95} onPress={onOpen}>
        {post.uri && post.mediaType === 'video' ? (
          <ReelVideo uri={post.uri} />
        ) : post.uri ? (
          <Image source={{ uri: post.uri }} style={styles.postImage} resizeMode="cover" />
        ) : (
          <View style={[styles.postImage, { backgroundColor: post.color || '#D8E7DF' }]}>
            <Text style={styles.postEmoji}>{post.emoji || '🌿'}</Text>
          </View>
        )}
      </TouchableOpacity>

      <View style={styles.actions}>
        <TouchableOpacity onPress={handleLike} disabled={interactionBusy}>
          <Text style={[styles.actionIcon, liked && styles.liked]}>{liked ? '♥' : '♡'}</Text>
        </TouchableOpacity>
        <TouchableOpacity onPress={openComments}>
          <Text style={styles.actionIcon}>▢</Text>
        </TouchableOpacity>
        <TouchableOpacity onPress={() => Alert.alert('Share', 'Sharing will be added next.')}>
          <Text style={styles.actionIcon}>➤</Text>
        </TouchableOpacity>
        <View style={styles.actionSpacer} />
        <TouchableOpacity onPress={() => setSaved(!saved)}>
          <Text style={styles.actionIcon}>{saved ? '▣' : '♧'}</Text>
        </TouchableOpacity>
      </View>

      <Text style={styles.likes}>{likeCount.toLocaleString()} likes</Text>
      {!!post.caption && (
        <Text style={styles.caption}>
          <Text style={styles.username}>{post.user || 'you'} </Text>{post.caption}
        </Text>
      )}
      <TouchableOpacity onPress={openComments}>
        <Text style={styles.comments}>View all comments</Text>
      </TouchableOpacity>
      {!!post.createdAt && <Text style={styles.time}>{formatPostTime(post.createdAt)}</Text>}

      <Modal
        visible={commentsOpen}
        animationType="slide"
        transparent
        onRequestClose={() => setCommentsOpen(false)}
      >
        <View style={{ flex: 1, justifyContent: 'flex-end', backgroundColor: 'rgba(0,0,0,0.35)' }}>
          <View style={{ backgroundColor: '#FFFFFF', borderTopLeftRadius: 18, borderTopRightRadius: 18, padding: 18, maxHeight: '75%' }}>
            <View style={{ flexDirection: 'row', justifyContent: 'space-between', alignItems: 'center', marginBottom: 12 }}>
              <Text style={{ fontSize: 18, fontWeight: '600', color: '#222222' }}>Comments</Text>
              <TouchableOpacity onPress={() => setCommentsOpen(false)}>
                <Text style={{ fontSize: 22, color: '#222222' }}>×</Text>
              </TouchableOpacity>
            </View>

            <ScrollView style={{ flexGrow: 0, maxHeight: 360 }}>
              {comments.length === 0 ? (
                <Text style={{ color: '#777777', paddingVertical: 12 }}>No comments yet.</Text>
              ) : comments.map(comment => (
                <View key={String(comment.id)} style={{ paddingVertical: 8 }}>
                  <Text style={{ color: '#222222' }}>
                    <Text style={{ fontWeight: '600' }}>{comment.username || 'you'} </Text>
                    {comment.text}
                  </Text>
                </View>
              ))}
            </ScrollView>

            <View style={{ flexDirection: 'row', alignItems: 'center', marginTop: 12 }}>
              <TextInput
                value={commentDraft}
                onChangeText={setCommentDraft}
                placeholder="Write a comment..."
                maxLength={2000}
                editable={!interactionBusy}
                style={{ flex: 1, borderWidth: 1, borderColor: '#DDDDDD', borderRadius: 20, paddingHorizontal: 14, paddingVertical: 9, color: '#222222' }}
              />
              <TouchableOpacity onPress={submitComment} disabled={interactionBusy || !commentDraft.trim()} style={{ padding: 10 }}>
                <Text style={{ color: '#0095F6', fontWeight: '600' }}>Post</Text>
              </TouchableOpacity>
            </View>
          </View>
        </View>
      </Modal>
    </View>
  );
}

            {(() => {

                <Story

                    uri: profile?.avatarUrl
                  }}
                  onPress={() => {
                    if (ownGroup) openStoryGroup(ownGroup.items);
                    else createStory();
                  }}
                />
              );
            })()}
            {groupedStories.filter(group => !group.own).map(group => (
              <Story
                key={group.key}
                item={{ ...group.items[0], name: group.name, own: false }}
                onPress={() => openStoryGroup(group.items)}

            ))}
            {stories.filter(s => !s.own).map(s => (
              <Story key={s.name} item={s} onPress={() => openStoryGroup([s])} />
            ))}
          </ScrollView>
          <View style={styles.feedDivider} />
          {feedPosts.map((p, i) => <PostCard key={p.id || `${p.user}-${i}`} post={{ ...p, avatarUrl: (p.user || 'you') === 'you' ? (profile?.avatarUrl || p.avatarUrl) : p.avatarUrl }} onOpen={() => setSelectedPost(p)} />)}

        <ScrollView style={styles.content} contentContainerStyle={{ padding: 20, paddingBottom: 36 }}>
          {!profile ? (

              प्रोफ़ाइल लोड हो रही है…
            </Text>
          ) : (
            <View>
              <View style={{ flexDirection: 'row', alignItems: 'center', marginBottom: 22 }}>
                <TouchableOpacity onPress={chooseProfilePhoto} disabled={profileBusy}>
                  {profile.avatarUrl ? (
                    <Image source={{ uri: profile.avatarUrl }}
                      onError={(e) => console.log('PROFILE_AVATAR_IMAGE_ERROR:', profile.avatarUrl, e.nativeEvent?.error)}
                      onLoad={() => console.log('PROFILE_AVATAR_IMAGE_LOADED:', profile.avatarUrl)}
                      style={{ width: 88, height: 88, borderRadius: 44, backgroundColor: '#DDE7F5' }} />

import * as FileSystem from 'expo-file-system/legacy';
import * as SecureStore from 'expo-secure-store';
const API_BASE_URL = 'http://127.0.0.1:5000';

const AUTH_TOKEN_KEY = 'socialapp_auth_token';
const AUTH_USERNAME_KEY = 'socialapp_auth_username';

export async function saveAuthSession(token, username) {
  await SecureStore.setItemAsync(AUTH_TOKEN_KEY, token);
  await SecureStore.setItemAsync(AUTH_USERNAME_KEY, username);
}

export async function clearAuthSession() {
  await SecureStore.deleteItemAsync(AUTH_TOKEN_KEY);
  await SecureStore.deleteItemAsync(AUTH_USERNAME_KEY);
}

export async function getAuthSession() {
  const token = await SecureStore.getItemAsync(AUTH_TOKEN_KEY);
  const username = await SecureStore.getItemAsync(AUTH_USERNAME_KEY);
  return token && username ? { token, username } : null;
}


export async function registerAccount(username, password, phoneNumber, email, fullName) {
  const response = await fetch(`${API_BASE_URL}/api/auth/register`, {
    method: 'POST',

    body: JSON.stringify({
      username: username.trim(),
      full_name: fullName.trim(),
      email: email.trim().toLowerCase(),
      password,
      phone_number: phoneNumber.trim(),
    }),
  });
  const data = await response.json();
  if (!response.ok) throw new Error(data.error || 'Registration failed.');
  await saveAuthSession(data.token, data.username);
  return data;
}

export async function verifyAuthSession() {
  const session = await getAuthSession();
  if (!session?.token) {
    return { valid: false, reason: 'missing_token' };
  }

  const response = await originalFetch(`${API_BASE_URL}/api/auth/me`, {
    method: 'GET',
    headers: { Authorization: `Bearer ${session.token}` },
  });

  let data = {};
  try {
    data = await response.json();
  } catch (_) {}

  return response.ok
    ? { valid: true, username: data.username }
    : { valid: false, status: response.status, reason: data.error || 'Token rejected' };
}

export async function logoutAccount() {
  try {
    await fetch(`${API_BASE_URL}/api/auth/logout`, { method: 'POST' });
  } finally {
    await clearAuthSession();
  }
}

const originalFetch = globalThis.fetch.bind(globalThis);
globalThis.fetch = async (input, init = {}) => {
  const headers = new Headers(init.headers || {});
  const token = await SecureStore.getItemAsync(AUTH_TOKEN_KEY);
  if (token) {
    headers.set('Authorization', `Bearer ${token}`);
  } else {
    headers.delete('Authorization');
  }
  const url = typeof input === 'string' ? input : input.url;
  const response = await originalFetch(input, { ...init, headers });
  if (String(url).includes('/api/')) {
    console.log('[AUTH DEBUG]', response.status, String(init.method || 'GET').toUpperCase(), String(url).split('?')[0], 'tokenSent=' + headers.has('Authorization'));
  }
  return response;
};

export async function uploadContent(type, asset, caption = '') {
  const extension = (
    asset.fileName || asset.uri || ''
  ).split('.').pop().split('?')[0].toLowerCase();

  const fallbackExtension = type === 'reels' ? 'mp4' : 'jpg';
  const fileExtension = /^[a-z0-9]+$/.test(extension)
    ? extension
    : fallbackExtension;

  const mimeTypes = {
    jpg: 'image/jpeg',
    jpeg: 'image/jpeg',
    png: 'image/png',
    webp: 'image/webp',
    gif: 'image/gif',
    mp4: 'video/mp4',
    mov: 'video/quicktime',
    webm: 'video/webm',
  };

  const authSession = await getAuthSession();
  const result = await FileSystem.uploadAsync(
    `${API_BASE_URL}/api/${type}`,
    asset.uri,
    {
      httpMethod: 'POST',
      headers: authSession ? { Authorization: `Bearer ${authSession.token}` } : {},

      uploadType: FileSystem.FileSystemUploadType.MULTIPART,
      fieldName: 'file',
      mimeType: mimeTypes[fileExtension] ||
        (asset.type === 'video' ? 'video/mp4' : 'image/jpeg'),
      parameters: {
        caption,
        username: authSession?.username || 'you',
      },
    }
  );

  let data;
  try {
    data = JSON.parse(result.body);
  } catch {
    throw new Error(`सर्वर का जवाब पढ़ा नहीं जा सका (HTTP ${result.status})`);
  }

  if (result.status < 200 || result.status >= 300) {
    throw new Error(data.error || `Upload failed: ${result.status}`);
  }

  const item = data.item;
  if (!item) {
    throw new Error('सर्वर ने अपलोड की जानकारी नहीं लौटाई।');
  }

  return {
    ...item,
    uri: item.mediaUrl ? `${API_BASE_URL}${item.mediaUrl}` : null,
    likes: '0',
  };
}

    `${API_BASE_URL}/api/posts/${encodeURIComponent(postId)}/comments`,
    {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ username, text }),
    }
  );
  const data = await response.json();
  if (!response.ok) {
    throw new Error(data.error || 'Could not create comment.');
  }
  return data.comment;
}

        return jsonify({"error": "Unknown content type"}), 404

    media = request.files.get("file")
    if media is None or not media.filename:
        return jsonify({"error": "Media file is required"}), 400

    original_name = secure_filename(media.filename)
    if "." not in original_name:
        return jsonify({"error": "File extension is missing"}), 400

    extension = original_name.rsplit(".", 1)[1].lower()
    if extension not in ALLOWED_EXTENSIONS[kind]:
        return jsonify({"error": "This file type is not allowed"}), 415

    caption = request.form.get("caption", "").strip()
    username = request.form.get("username", "you").strip() or "you"

    stored_name = f"{uuid.uuid4().hex}.{extension}"
    destination = folder / stored_name

    try:
        media.save(destination)

        if destination.stat().st_size == 0:
            destination.unlink(missing_ok=True)
            return jsonify({"error": "The uploaded file is empty"}), 400

        now = datetime.now(timezone.utc)
        created_at = now.isoformat()
        expires_at = (
            (now + timedelta(hours=24)).isoformat()
            if kind == "stories" else None
        )
        media_type = (
            "video" if extension in {"mp4", "mov", "webm"} else "image"
        )
        media_url = f"/uploads/{kind}/{stored_name}"

        with get_db() as db:
            cursor = db.execute(
                """
                INSERT INTO content
                    (kind, username, caption, media_url, media_type,
                     created_at, expires_at)
                VALUES (?, ?, ?, ?, ?, ?, ?)
                """,
                (
                    kind, username, caption, media_url, media_type,
                    created_at, expires_at,
                ),
            )
            row = db.execute(
                "SELECT * FROM content WHERE id=?",
                (cursor.lastrowid,),
            ).fetchone()

        return jsonify({"ok": True, "item": content_json(row)}), 201

    except Exception:
        destination.unlink(missing_ok=True)
        app.logger.exception("Could not save uploaded media")
        return jsonify({"error": "Could not save media"}), 500

def update_profile(username):
    username = username.strip().lstrip("@")
    data = request.get_json(silent=True) or {}
    # This basic API uses the supplied username as identity. Add authenticated
    # sessions before exposing it to untrusted users.
    display_name = str(data.get("displayName", "")).strip()
    bio = str(data.get("bio", "")).strip()
    if len(display_name) > 60 or len(bio) > 300:
        return jsonify({"error": "Name max 60 and bio max 300 characters"}), 400
    now = datetime.now(timezone.utc)
    with get_db() as db:
        profile_json(db, username)
        row = db.execute(
            "SELECT * FROM profiles WHERE username=?", (username,)
        ).fetchone()
        if "displayName" in data and display_name != row["display_name"]:
            previous = row["name_changed_at"]
            if previous:
                try:
                    last = datetime.fromisoformat(previous)
                    if now - last < timedelta(days=90):
                        next_date = (last + timedelta(days=90)).date().isoformat()
                        return jsonify({
                            "error": f"Display name अगली बार {next_date} से बदल सकते हैं।"
                        }), 429
                except ValueError:
                    pass
            db.execute(
                "UPDATE profiles SET display_name=?, name_changed_at=? WHERE username=?",
                (display_name or username, now.isoformat(), username),
            )
        if "bio" in data:
            db.execute("UPDATE profiles SET bio=? WHERE username=?", (bio, username))
        # Username is intentionally never accepted as an editable field.
        result = profile_json(db, username)
    return jsonify({"ok": True, "profile": result})

        """)

def _auth_user():
    header = request.headers.get("Authorization", "")
    if not header.startswith("Bearer "):
        return None
    token = header[7:].strip()
    if len(token) < 32:
        return None
    digest = hashlib.sha256(token.encode("utf-8")).hexdigest()
    now = datetime.now(timezone.utc).isoformat()
    with get_db() as db:
        row = db.execute("""
            SELECT s.username FROM auth_sessions s
            JOIN users u ON u.username=s.username
            WHERE s.token_hash=? AND s.expires_at>?
        """, (digest, now)).fetchone()
    return row["username"] if row else None

@app.before_request
def enforce_authenticated_mutations():
    if request.method in ("GET", "HEAD", "OPTIONS"):
        return None
    if request.path in (
        "/api/auth/register",
        "/api/auth/login",
        "/api/auth/forgot-password/request",
        "/api/auth/forgot-password/verify",
        "/api/auth/forgot-password/reset",
    ):
        return None
    if not request.path.startswith("/api/"):
        return None

@app.post("/api/auth/register")
def auth_register():
    data = request.get_json(silent=True) or {}
    email = str(data.get("email", "")).strip().lower()
    password = str(data.get("password", ""))
    phone_number = str(data.get("phone_number", "")).strip()
    phone_digits = "".join(ch for ch in phone_number if ch.isdigit())

    if "@" not in email or "." not in email.rsplit("@", 1)[-1] or len(email) > 254:
        return jsonify({"error": "कृपया सही ईमेल आईडी दर्ज करें।"}), 400
    if len(phone_digits) < 10 or len(phone_digits) > 15:
        return jsonify({"error": "कृपया सही मोबाइल नंबर दर्ज करें।"}), 400
    if len(password) < 4 or len(password) > 24:
        return jsonify({"error": "पासवर्ड 4 से 24 अक्षरों का रखें।"}), 400

    username = str(data.get("username", "")).strip().lstrip("@")
    full_name = str(data.get("full_name", "")).strip()

    if not (3 <= len(username) <= 30) or any(
        not (ch.isascii() and (ch.isalnum() or ch in "_."))
        for ch in username
    ):
        return jsonify({"error": "Username must be 3 to 30 letters, numbers, dots, or underscores."}), 400
    if not full_name or len(full_name) > 60:
        return jsonify({"error": "Full name is required and must be at most 60 characters."}), 400

    now = datetime.now(timezone.utc)
    token = secrets.token_urlsafe(32)
    digest = hashlib.sha256(token.encode("utf-8")).hexdigest()
    expires = (now + timedelta(days=30)).isoformat()

    try:
        with get_db() as db:
            if db.execute(
                "SELECT 1 FROM users WHERE username=? COLLATE NOCASE", (username,)
            ).fetchone():
                return jsonify({"error": "यह username पहले से इस्तेमाल हो रहा है।"}), 409

            db.execute(
                "INSERT INTO users(username,password_hash,created_at,phone_number,email) VALUES(?,?,?,?,?)",
                (username, generate_password_hash(password), now.isoformat(), phone_number, email)
            )
            db.execute(
                "INSERT INTO auth_sessions(token_hash,username,created_at,expires_at) VALUES(?,?,?,?)",
                (digest, username, now.isoformat(), expires)
            )
            profile_json(db, username)
            db.execute(
                "UPDATE profiles SET display_name=? WHERE username=?",
                (full_name, username)
            )
            db.execute(
                "INSERT OR IGNORE INTO user_settings(username,updated_at) VALUES(?,?)",
                (username, now.isoformat())
            )
    except sqlite3.IntegrityError:
        return jsonify({"error": "यह ईमेल पहले से किसी अकाउंट में इस्तेमाल हो रहा है।"}), 409

    return jsonify({"ok": True, "username": username, "token": token}), 201

@app.post("/api/auth/change-password")
def auth_change_password():
    username = getattr(g, "auth_username", None)
    if not username:
        return jsonify({"error": "Please log in again."}), 401

    data = request.get_json(silent=True) or {}
    old_password = str(data.get("old_password", ""))
    new_password = str(data.get("new_password", ""))
    confirm_password = str(data.get("confirm_password", ""))

    if not old_password or not new_password or not confirm_password:
        return jsonify({"error": "All password fields are required."}), 400
    if new_password != confirm_password:
        return jsonify({"error": "New password and confirmation do not match."}), 400
    if len(new_password) < 4 or len(new_password) > 24:
        return jsonify({"error": "New password must be 4 to 24 characters."}), 400
    if new_password == old_password:
        return jsonify({"error": "Choose a password different from the old password."}), 400

    with get_db() as db:
        row = db.execute(
            "SELECT password_hash FROM users WHERE username=?",
            (username,)
        ).fetchone()

        if not row or not check_password_hash(row["password_hash"], old_password):
            return jsonify({"error": "Old password is incorrect."}), 401

        db.execute(
            "UPDATE users SET password_hash=? WHERE username=?",
            (generate_password_hash(new_password), username)
        )

        header = request.headers.get("Authorization", "")
        token = header[7:].strip() if header.startswith("Bearer ") else ""
        current_digest = hashlib.sha256(token.encode("utf-8")).hexdigest() if token else ""
        db.execute(
            "DELETE FROM auth_sessions WHERE username=? AND token_hash!=?",
            (username, current_digest)
        )

    return jsonify({"ok": True, "message": "Password changed successfully."})




def _send_password_reset_email(email, otp):
    api_key = os.environ.get("RESEND_API_KEY", "").strip()
    sender = os.environ.get("RESEND_FROM_EMAIL", "").strip()
    if not api_key or not sender:
        raise RuntimeError("Email provider is not configured.")

    payload = {
        "from": sender,
        "to": [email],
        "subject": "Your SocialApp password reset code",
        "text": (
            f"Your SocialApp password reset code is {otp}.\n\n"
            "This code expires in 10 minutes. If you did not request this, "
            "you can ignore this email."
        ),
    }
    req = urllib.request.Request(
        "https://api.resend.com/emails",
        data=json.dumps(payload).encode("utf-8"),
        headers={
            "Authorization": f"Bearer {api_key}",
            "Content-Type": "application/json",
        },
        method="POST",
    )
    with urllib.request.urlopen(req, timeout=15) as response:
        if response.status < 200 or response.status >= 300:
            raise RuntimeError("Email provider rejected the request.")


@app.post("/api/auth/forgot-password/request")
def auth_forgot_password_request():
    data = request.get_json(silent=True) or {}
    email = str(data.get("email", "")).strip().lower()
    if len(email) > 254 or "@" not in email or "." not in email.rsplit("@", 1)[-1]:
        return jsonify({"error": "Enter a valid email address."}), 400

    api_key = os.environ.get("RESEND_API_KEY", "").strip()
    sender = os.environ.get("RESEND_FROM_EMAIL", "").strip()
    if not api_key or not sender:
        return jsonify({"error": "Password recovery email is not configured on the server yet."}), 503

    now = datetime.now(timezone.utc)
    now_iso = now.isoformat()
    with get_db() as db:
        user = db.execute(
            "SELECT username FROM users WHERE email=? COLLATE NOCASE", (email,)
        ).fetchone()
        if not user:
            return jsonify({
                "ok": True,
                "message": "If an account uses that email, a reset code will be sent."
            }), 200

        previous = db.execute(
            "SELECT last_sent_at FROM password_reset_otps WHERE email=? COLLATE NOCASE",
            (email,),
        ).fetchone()
        if previous:
            try:
                last_sent = datetime.fromisoformat(previous["last_sent_at"])
                if (now - last_sent).total_seconds() < 60:
                    return jsonify({
                        "ok": True,
                        "message": "If an account uses that email, a reset code will be sent."
                    }), 200
            except (TypeError, ValueError):
                pass

    otp = f"{secrets.randbelow(1000000):06d}"
    otp_hash = hmac.new(
        api_key.encode("utf-8"),
        f"{email}:{otp}".encode("utf-8"),
        hashlib.sha256,
    ).hexdigest()
    expires = (now + timedelta(minutes=10)).isoformat()

    try:
        _send_password_reset_email(email, otp)
    except (urllib.error.URLError, TimeoutError, RuntimeError, ValueError):
        app.logger.warning("Password reset email delivery failed.")
        return jsonify({"error": "Could not send the reset email. Check the server email configuration."}), 503

    with get_db() as db:
        db.execute("""
            INSERT INTO password_reset_otps
                (email, otp_hash, expires_at, last_sent_at, attempts, reset_token_hash, reset_expires_at)
            VALUES (?, ?, ?, ?, 0, NULL, NULL)
            ON CONFLICT(email) DO UPDATE SET
                otp_hash=excluded.otp_hash,
                expires_at=excluded.expires_at,
                last_sent_at=excluded.last_sent_at,
                attempts=0,
                reset_token_hash=NULL,
                reset_expires_at=NULL
        """, (email, otp_hash, expires, now_iso))

    return jsonify({
        "ok": True,
        "message": "If an account uses that email, a reset code will be sent."
    }), 200


@app.post("/api/auth/forgot-password/verify")
def auth_forgot_password_verify():
    data = request.get_json(silent=True) or {}
    email = str(data.get("email", "")).strip().lower()
    otp = str(data.get("otp", "")).strip()
    api_key = os.environ.get("RESEND_API_KEY", "").strip()

    if not api_key:
        return jsonify({"error": "Password recovery email is not configured on the server yet."}), 503
    if not email or not re.fullmatch(r"[0-9]{6}", otp):
        return jsonify({"error": "Enter the email address and 6-digit code."}), 400

    now = datetime.now(timezone.utc)
    with get_db() as db:
        row = db.execute(
            "SELECT * FROM password_reset_otps WHERE email=? COLLATE NOCASE", (email,)
        ).fetchone()
        if not row:
            return jsonify({"error": "The code is invalid or expired."}), 400
        if row["attempts"] >= 5:
            return jsonify({"error": "Too many attempts. Request a new code."}), 429
        try:
            expired = datetime.fromisoformat(row["expires_at"]) <= now
        except (TypeError, ValueError):
            expired = True
        if expired:
            return jsonify({"error": "The code is invalid or expired. Request a new code."}), 400

        submitted_hash = hmac.new(
            api_key.encode("utf-8"),
            f"{email}:{otp}".encode("utf-8"),
            hashlib.sha256,
        ).hexdigest()
        if not hmac.compare_digest(submitted_hash, row["otp_hash"]):
            db.execute(
                "UPDATE password_reset_otps SET attempts=attempts+1 WHERE email=? COLLATE NOCASE",
                (email,),
            )
            return jsonify({"error": "The code is invalid or expired."}), 400

        reset_token = secrets.token_urlsafe(32)
        reset_digest = hashlib.sha256(reset_token.encode("utf-8")).hexdigest()
        reset_expires = (now + timedelta(minutes=10)).isoformat()
        db.execute("""
            UPDATE password_reset_otps
            SET reset_token_hash=?, reset_expires_at=?
            WHERE email=? COLLATE NOCASE
        """, (reset_digest, reset_expires, email))

    return jsonify({"ok": True, "reset_token": reset_token}), 200


@app.post("/api/auth/forgot-password/reset")
def auth_forgot_password_reset():
    data = request.get_json(silent=True) or {}
    email = str(data.get("email", "")).strip().lower()
    reset_token = str(data.get("reset_token", ""))
    new_password = str(data.get("new_password", ""))
    confirm_password = str(data.get("confirm_password", ""))

    if not email or not reset_token:
        return jsonify({"error": "Verify the email code first."}), 400
    if new_password != confirm_password:
        return jsonify({"error": "Passwords do not match."}), 400
    if len(new_password) < 4 or len(new_password) > 24:
        return jsonify({"error": "Password must be 4 to 24 characters long."}), 400

    reset_digest = hashlib.sha256(reset_token.encode("utf-8")).hexdigest()
    now = datetime.now(timezone.utc)
    with get_db() as db:
        row = db.execute(
            "SELECT reset_token_hash, reset_expires_at FROM password_reset_otps WHERE email=? COLLATE NOCASE",
            (email,),
        ).fetchone()
        if not row or not row["reset_token_hash"] or not hmac.compare_digest(
            reset_digest, row["reset_token_hash"]
        ):
            return jsonify({"error": "The reset session is invalid. Verify the code again."}), 400
        try:
            if not row["reset_expires_at"] or datetime.fromisoformat(row["reset_expires_at"]) <= now:
                return jsonify({"error": "The reset session expired. Verify the code again."}), 400
        except (TypeError, ValueError):
            return jsonify({"error": "The reset session expired. Verify the code again."}), 400

        user = db.execute(
            "SELECT username FROM users WHERE email=? COLLATE NOCASE", (email,)
        ).fetchone()
        if not user:
            return jsonify({"error": "The account could not be found."}), 400

        db.execute(
            "UPDATE users SET password_hash=? WHERE username=?",
            (generate_password_hash(new_password), user["username"]),
        )
        db.execute(
            "DELETE FROM auth_sessions WHERE username=?", (user["username"],)
        )
        db.execute(
            "DELETE FROM password_reset_otps WHERE email=? COLLATE NOCASE", (email,)
        )

    return jsonify({"ok": True, "message": "Password reset successfully. Please log in again."}), 200
